<?php
// Read a compressed event log, one record at a time.

function events(string $file): Generator
{
    $stream = new SplFileObject("compress.zlib://$file");

    foreach ($stream as $line) {
        if (trim($line) !== '') {
            yield json_decode($line, flags: JSON_THROW_ON_ERROR);
        }
    }
}

foreach (events('events.ndjson.gz') as $event) {
    echo "{$event->type}\n";
}

<?php
// Extract headlines from HTML with CSS selectors.

$html = <<<'HTML'
    <!doctype html>
    <article><h2>Streaming data</h2></article>
    <article><h2>Building APIs</h2></article>
    HTML;

$page = Dom\HTMLDocument::createFromString($html);
$headlines = [];

foreach ($page->querySelectorAll('article h2') as $heading) {
    $headlines[] = $heading->textContent;
}

echo json_encode($headlines, JSON_PRETTY_PRINT);

<?php
// Sign a message, then detect whether it has changed.

$keys = sodium_crypto_sign_keypair();
$secret = sodium_crypto_sign_secretkey($keys);
$public = sodium_crypto_sign_publickey($keys);
$message = 'Release approved.';

$signature = sodium_crypto_sign_detached($message, $secret);

var_dump(sodium_crypto_sign_verify_detached(
    $signature, $message, $public
));

var_dump(sodium_crypto_sign_verify_detached(
    $signature, 'Release changed.', $public
));

<?php
// Map a web request to a JSON response.

class UserController
{
    public function __construct(
        private readonly UserRepository $users,
    ) {}

    #[Route('/users/{id}', methods: ['GET'])]
    public function show(int $id): Response
    {
        $user = $this->users->find($id)
            ?? throw new NotFoundException();

        return Response::json([
            'id' => $user->id,
            'name' => $user->name,
        ]);
    }
}

PHP 8.5 adds a built-in URI extension to parse, normalize, and handle URLs following RFC 3986 and WHATWG URL standards.
